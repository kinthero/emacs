** file

#+begin_src emacs-lisp
  (my/leader
    "ff" #'find-file
    "fc" #'my/open-config
    "fC" #'my/reload-emacs-config
    "fd" #'dired
    "fw" #'save-buffer
    "fW" #'save-some-buffers)

  (defconst my/emacs-config-org
    (expand-file-name "config.org"
  		    user-emacs-directory))

  (defconst my/emacs-config-el
    (expand-file-name "config.el"
                      user-emacs-directory))

  (defun my/open-config ()
    (interactive)
    (find-file my/emacs-config-org))

  (defun my/reload-emacs-config ()
    (interactive)
    (when-let ((buffer (get-file-buffer my/emacs-config-org)))
      (with-current-buffer buffer
        (save-buffer)))
    (org-babel-tangle-file my/emacs-config-org)
    (load-file my/emacs-config-el)
    (message "Emacs config reloaded!"))
#+end_src

*** 切换透明/不透明

#+begin_src emacs-lisp
  (defconst my/default-alpha 80)
  (defun my/toggle-alpha ()
    "Toggle frame opacity."
    (interactive)
    (let ((alpha (frame-parameter nil 'alpha)))
      (set-frame-parameter
       nil
       'alpha
       (if (or (null alpha)
               (= (if (consp alpha)
                      (car alpha)
                    alpha)
                  100))
           (cons my/default-alpha 100)
         '(100 . 100)))))
#+end_src

  (use-package valign
    :ensure t)

  (add-hook 'org-mode-hook #'valign-mode)

  ;; Fix valign + org-indent after folding/unfolding Org headings.
  (with-eval-after-load 'valign
    (defun my-valign-org-cycle-refresh (&rest _)
      (when (and (derived-mode-p 'org-mode)
                 (bound-and-true-p valign-mode))
        ;; Wait until org-cycle/org-indent has finished updating
        ;; the visibility and indentation, then refontify.
        (run-at-time
         0 nil
         (lambda (buffer)
           (when (buffer-live-p buffer)
             (with-current-buffer buffer
               (font-lock-flush))))
         (current-buffer))))

    (add-hook 'org-cycle-hook
              #'my-valign-org-cycle-refresh
              100))
#+end_src
